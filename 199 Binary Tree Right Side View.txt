Intuition
We use level - order technique to solve this questions.
Approach
Step 1 : Travel the tree in the level order and store each level in list. (List of list)
Step 2 : In the answer list add last element of each level.
Step 3 : Return answer list
Complexity
Time complexity:
O(n) to travel all nodes
Space complexity:
O(n) to store elements in the list.
Code
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<Integer> rightSideView(TreeNode root) 
    {
        List<Integer> ans = new ArrayList<>();
        if(root == null)
        {
            return ans;
        }

        Queue<TreeNode> q = new LinkedList<>();
        List<List<Integer>> LL = new ArrayList<>();
        List<Integer> L = new ArrayList<>();

        q.add(root);
        q.add(null);

        while(!q.isEmpty())
        {
            TreeNode T = q.remove();

            if(T != null)
            {
                L.add(T.val);
                if(T.left != null)
                {
                    q.add(T.left);
                }
                if(T.right != null)
                {
                    q.add(T.right);
                }
            }
            else
            {
                LL.add(new ArrayList<>(L));
                L.clear();

                if(!q.isEmpty())
                {
                    q.add(null);
                }
            }
        }

        for(List<Integer> S : LL)
        {
            ans.add(S.get(S.size()-1));
        }

        return ans;

    }
}
