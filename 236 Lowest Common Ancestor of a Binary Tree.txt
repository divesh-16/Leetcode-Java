Approach
Any node which occurs to be 'p' or 'q' first is to be returned.
If node is not p and q, but its children are p and q - then return this node.
Otherwise return either left or right child once traveled down.
Complexity
Time complexity:
O(n) to travel all nodes.
Space complexity:
Code
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode(int x) { val = x; }
 * }
 */
class Solution 
{
    public TreeNode search(TreeNode root,TreeNode p, TreeNode q)
    {
        if(root == null)
        {
            return null;
        }
        else if(root.val == p.val || root.val == q.val)
        {
            return root;
        }
        else
        {
            TreeNode lc = search(root.left,p,q);
            TreeNode rc = search(root.right,p,q);

            if(lc != null && rc != null)
            {   
                return root;    
            }
            else
            {
                return (lc != null ? lc : rc);
            }

        }
    }

    public TreeNode lowestCommonAncestor(TreeNode root, TreeNode p, TreeNode q) 
    {     
        // Approach 1   
        //return search(root,p,q);

        // Approach 2

        if(root == null)
        {
            return null;
        }

        if(root == p || root == q)
        {
            return root;
        }

        TreeNode left = lowestCommonAncestor(root.left,p,q);
        TreeNode right = lowestCommonAncestor(root.right,p,q);

        if(left != null && right != null)
        {
            return root;
        }
        else
        {
            return (left != null ? left : right);
        }
    }
}
